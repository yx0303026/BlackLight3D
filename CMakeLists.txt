cmake_minimum_required(VERSION 3.25)

project(BlackLight3D LANGUAGES CXX)

option(BLACKLIGHT_FETCH_DEPENDENCIES "Fetch pinned dependencies when packages are unavailable" ON)
option(BLACKLIGHT_BUILD_TESTS "Build BlackLight UI tests" ON)

set(BLACKLIGHT_ARCH "x64" CACHE STRING "Target architecture used in the build output path")
set_property(CACHE BLACKLIGHT_ARCH PROPERTY STRINGS x64 x86)
if(NOT BLACKLIGHT_ARCH STREQUAL "x64" AND NOT BLACKLIGHT_ARCH STREQUAL "x86")
	message(FATAL_ERROR "BLACKLIGHT_ARCH must be either x64 or x86.")
endif()

# Keep every generated artifact in the architecture/configuration-specific output directory.
set(BLACKLIGHT_OUTPUT_DIRECTORY
	"${CMAKE_SOURCE_DIR}/build/${BLACKLIGHT_ARCH}/$<LOWER_CASE:$<CONFIG>>"
)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${BLACKLIGHT_OUTPUT_DIRECTORY}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${BLACKLIGHT_OUTPUT_DIRECTORY}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${BLACKLIGHT_OUTPUT_DIRECTORY}")
foreach(BLACKLIGHT_CONFIGURATION Debug Release RelWithDebInfo MinSizeRel)
	string(TOLOWER "${BLACKLIGHT_CONFIGURATION}" BLACKLIGHT_CONFIGURATION_LOWER)
	set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${BLACKLIGHT_CONFIGURATION}
		"${CMAKE_SOURCE_DIR}/build/${BLACKLIGHT_ARCH}/${BLACKLIGHT_CONFIGURATION_LOWER}"
	)
	set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${BLACKLIGHT_CONFIGURATION}
		"${CMAKE_SOURCE_DIR}/build/${BLACKLIGHT_ARCH}/${BLACKLIGHT_CONFIGURATION_LOWER}"
	)
	set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${BLACKLIGHT_CONFIGURATION}
		"${CMAKE_SOURCE_DIR}/build/${BLACKLIGHT_ARCH}/${BLACKLIGHT_CONFIGURATION_LOWER}"
	)
endforeach()

if(BLACKLIGHT_BUILD_TESTS)
	enable_testing()
endif()

add_subdirectory(BlackLight)
add_subdirectory(Demo)
add_subdirectory(BlackLightEditor)
